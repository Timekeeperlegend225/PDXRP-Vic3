# GORA UI+ — State Assets: nationally-owned level counter (staircase)
#
# Building scope. Counts how many levels of this building are owned by ROOT.
# Result is written to ROOT.temp_bldg_nat_levels.
#
# Why a staircase? Victoria 3's `levels_owned_by_country` is a value-comparison
# trigger, not a bare value, so we cannot read it directly into a variable.
# We instead test "value >= N" for each N from 1..200 and increment a counter,
# which produces the same numeric result. 200 covers every realistic building
# level (vanilla caps far below that; even with conversion mods 200 is generous).
#
# This is the hottest single piece of code in the State Assets system, so we
# parameterise the per-step logic into a tiny helper effect and call it
# 200 times. Editing the step body in one place stays trivial.

gora_sa_staircase_step_effect = {
	if = {
		limit = { levels_owned_by_country = { target = ROOT value >= $n$ } }
		ROOT = { change_variable = { name = temp_bldg_nat_levels add = 1 } }
	}
}

# ─── Per-building national value (cost tier × 250 × levels) ───────────────
# Building scope. Computes ROOT's nationally-owned value for THIS building.
# Reads ROOT.temp_bldg_nat_levels (must be set first by gora_sa_staircase_nat_levels_effect).
# Result is written to ROOT.temp_bldg_nat_value.
gora_sa_compute_nat_value_for_building_effect = {
	ROOT = { set_variable = { name = temp_bldg_nat_value value = 0 } }

	if = {
		limit = { is_building_group = bg_trade }
		ROOT = { set_variable = { name = temp_bldg_nat_value value = 100 } }
	}
	else_if = {
		limit = {
			OR = {
				is_building_group = bg_agriculture
				is_building_group = bg_ranching
				is_building_group = bg_plantations
				is_building_group = bg_logging
				is_building_group = bg_rubber
				is_building_group = bg_fishing
				is_building_group = bg_whaling
			}
		}
		ROOT = { set_variable = { name = temp_bldg_nat_value value = 200 } }
	}
	else_if = {
		# Railway must precede bg_private_infrastructure medium tier.
		limit = { is_building_type = building_railway }
		ROOT = { set_variable = { name = temp_bldg_nat_value value = 800 } }
	}
	else_if = {
		limit = {
			OR = {
				is_building_group = bg_mining
				is_building_group = bg_oil_extraction
				is_building_group = bg_private_infrastructure
				is_building_group = bg_arts
				is_building_group = bg_power
			}
		}
		ROOT = { set_variable = { name = temp_bldg_nat_value value = 400 } }
	}
	else_if = {
		limit = { is_building_group = bg_light_industry }
		ROOT = { set_variable = { name = temp_bldg_nat_value value = 600 } }
	}
	else = {
		# Heavy industry, military industry, anything else.
		ROOT = { set_variable = { name = temp_bldg_nat_value value = 800 } }
	}

	# value = tier * PRIVATIZATION_PER_LEVEL_COST(250) * levels
	ROOT = {
		change_variable = { name = temp_bldg_nat_value multiply = 250 }
		change_variable = { name = temp_bldg_nat_value multiply = var:temp_bldg_nat_levels }
	}
}
