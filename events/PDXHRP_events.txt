namespace = crash

crash.1 = { #For Bodi
    type = country_event
	placement = root
	
    title = "Crash"
    description = "Nooooo !!!!"
	flavor = "Something Happen"
	
    event_image = { 
		video = unspecific_politicians_arguing
	}

	icon = "gfx/interface/icons/alert_icons/bankruptcy.dds"

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	immediate = {
		set_global_variable = {
			name = national_integration_var_b
			value = var:national_integration_var
		}
		set_variable = {
			name = IP_value
			value = investment_pool
		}
		set_variable = {
			name = crash_radicals_var
			value = var:national_integration_var
		}
		change_variable = {
			name = crash_radicals_var
			multiply = 0.10
		}
		change_variable = {
			name = IP_value
			multiply = -1
		}
		change_variable = {
			name = IP_value
			multiply = var:national_integration_var
		}
		add_investment_pool = var:IP_value
		add_modifier = {
			name = HRP_crash           
			months = 120
			multiplier = var:national_integration_var
			is_decaying = yes
		}
		add_radicals = {
			value = var:crash_radicals_var
		}
		every_scope_building = {
			set_global_variable = {
				name = building_crash_var
				value = private_ownership_fraction
			}
			add_modifier = {
				name = HRP_building_crash           
				months = 120
				multiplier = {
					add = global_var:building_crash_var
					multiply = global_var:national_integration_var_b
				}
				is_decaying = yes
			}	
		}
	}
	
	duration = 3
	
    option = {
		name = "All is lost"
		show_as_tooltip = {
		add_investment_pool = var:IP_value
			add_modifier = {
				name = HRP_crash           
				months = 120
				multiplier = var:national_integration_var
				is_decaying = yes
			}
			add_radicals = {
				value = var:crash_radicals_var
			}
		}
		custom_tooltip ={
			text = "Buildings will also get up to -70% throuput bonus depending of your integration and their private ownership ratio"
		}
	}
}
