namespace = gora_agdp

# ════════════════════════════════════════════════════════════════════════════
# GORA UI+ — Advanced GDP Tracker: events
# ════════════════════════════════════════════════════════════════════════════
# .1 — Annual: roll before_gdp/current_gdp and recompute the Jan-to-Jan
#      gdp_growth_rate.
# .3 — Monthly: roll the 12-slot snapshot table and recompute all derived
#      stats. Must be called BEFORE the monthly record is overwritten.
# ════════════════════════════════════════════════════════════════════════════

gora_agdp.1 = {
	type = country_event
	hidden = yes
	trigger = {}

	immediate = {
		if = {
			limit = { has_variable = current_gdp }
			set_variable = { name = before_gdp value = var:current_gdp }
		}
		else = {
			set_variable = { name = before_gdp value = gdp }
		}
		set_variable = { name = current_gdp value = gdp }

		# Compute once and cache; the locale tooltip reads the variable directly.
		set_variable = {
			name = gdp_growth_rate
			value = {
				add      = gdp
				subtract = var:before_gdp
				divide   = { value = var:before_gdp min = 1 }
			}
		}
	}
}

gora_agdp.3 = {
	type = country_event
	hidden = yes
	trigger = {}

	immediate = {
		# Order matters: rolling diff is computed against the snapshot we are
		# about to overwrite, so it MUST run first.
		GDPGR_update_rolling_gdp_growth    = yes
		GDPGR_update_monthly_record        = yes
		# Then refresh the derived stats.
		GDPGR_update_past_month_stats      = yes
		GDPGR_calculate_past_year_averages = yes
	}
}
