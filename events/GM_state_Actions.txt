namespace = gm_state_actions

gm_state_actions.1 = {
	type = country_event
	placement = ROOT
	title = gm_state_actions.1.t
	desc = gm_state_actions.1.u

	event_image = {
		texture = "gfx/event_pictures/void.dds"
	} 

	duration = 1

	immediate = {
		random_country = {
			limit = {
				has_modifier = GM_target
			}
			save_scope_as = scope_GM_target
		}
	}
	
	trigger = {
		#triggered by diplo action
	}
	
	#GM options
	
	option = {
		name = gm_state_actions.1.a
		default_option = yes
		remove_global_variable = GM_taking_action
		scope:scope_target_state = {
			remove_modifier = GM_working_on_state_mod
		}
	}
	
	option = {
		name = gm_state_actions.1.b
		trigger = {
			scope:scope_GM_target = {
				NOT = {
					has_claim = scope:scope_target_state 
				}
			}
		}
		
		scope:scope_target_state = {
			state_region = {
				add_claim = scope:scope_GM_target
			}
		}
		trigger_event = {
			id = gm_state_actions.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = gm_state_actions.1.c
		trigger = {
			scope:scope_GM_target = {
				has_claim = scope:scope_target_state 
			}
		}
		
		scope:scope_target_state = {
			state_region = {
				remove_claim = scope:scope_GM_target
			}
		}
		trigger_event = {
			id = gm_state_actions.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = gm_state_actions.1.d
		trigger = {
			scope:scope_target_state  = {
				is_incorporated = no
			}
		}
		
		scope:scope_target_state = {
			set_state_type = incorporated
		}
		trigger_event = {
			id = gm_state_actions.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = gm_state_actions.1.e
		trigger = {
			scope:scope_target_state  = {
				is_incorporated = yes
			}
		}
		
		scope:scope_target_state = {
			set_state_type = unincorporated
		}
		trigger_event = {
			id = gm_state_actions.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = gm_state_actions.1.f
		trigger = {
			scope:scope_target_state = {
				NOT = {
					owner = scope:scope_GM_target
				}
			}
		}
		scope:scope_target_state = {
			set_state_owner = scope:scope_GM_target
		}
		trigger_event = {
			id = gm_state_actions.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = gm_state_actions.1.g
		trigger = {
			scope:scope_target_state  = {
				has_modifier = recently_conquered_state
			}
		}
		remove_global_variable = GM_taking_action
		scope:scope_target_state = {
			remove_modifier = recently_conquered_state
			remove_modifier = GM_working_on_state_mod
		}
	}
	
	# Loyalist, Radicals
	option = {
		name = value_selection.4100.t
		trigger_event = {
			id = value_selection.4100
			days = 0
			popup = yes
		}
	}
	
	# Devastation
	option = {
		name = gm_state_actions.3.t
		trigger_event = {
			id = gm_state_actions.3
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = gm_state_actions.1.h
		remove_global_variable = GM_taking_action
		scope:scope_target_state = {
			owner = {
				save_scope_as = scope_origin_country
				add_modifier = {
					name = test1
				}
				create_country = {
					origin = scope:scope_origin_country
					state = scope:scope_target_state
					on_created = {
						activate_law = law_type:law_universal_suffrage
						activate_law = law_type:law_parliamentary_republic
						activate_law = law_type:law_national_militia
						add_modifier = {
							name = test2
						}
					}
				}
			}
			random_country = {
				limit = {
					has_modifier = test1
				}
				save_scope_as = scope_origin_country
			}
			random_country = {
				limit = {
					has_modifier = test2
				}
				save_scope_as = scope_revolter
				create_diplomatic_play = {
					target_country = scope:scope_origin_country
					type = dp_regime_change
					war = yes
					add_war_goal = {
						holder = scope:scope_origin_country
						target_country = scope:scope_revolter
						type = annex_country
					}
				}
			}
			every_country = {
				limit = {
					has_modifier = test1
				}
				remove_modifier = test1
			}
			every_country = {
				limit = {
					has_modifier = test2
				}
				remove_modifier = test2
			}
			remove_modifier = GM_working_on_state_mod
		}
	}
}
