value_selection.4 = {
	type = country_event
	placement = ROOT
	title = value_selection.4.t
	desc = value_selection.4.u

	event_image = {
		texture = "gfx/event_pictures/void.dds"
	} 

	duration = 1
	
	trigger = {
		#triggered by diplo action
	}

	immediate = {
		random_country = {
			limit = {
				has_modifier = GM_interacted_mod
			}
			save_scope_as = GM_interacted_scope
		}
		random_country = {
			limit = {
				has_modifier = GM_target
			}
			save_scope_as = scope_GM_target
		}
		scope:GM_interacted_scope = {
			if = {
				limit = {
					has_war_with = scope:scope_GM_target
				}
				random_scope_war = {
					limit = {
						any_war_participant =  { 
							this = scope:scope_GM_target
						}
					}
					save_scope_as = scope_GM_war
				}
			}
			else = {
				random_scope_war = {
					save_scope_as = scope_GM_war
				}
			}
		}
	}

		name = value_selection.4.a
