namespace = value_selection

value_selection.1 = {
	type = country_event
	placement = ROOT
	title = value_selection.1.t
	desc = value_selection.1.u

	event_image = {
		texture = "gfx/event_pictures/void.dds"
	} 

	duration = 1
	
	trigger = {
		#triggered by diplo action
	}

	immediate = {
		random_country = {
			limit = {
				has_modifier = GM_interacted_mod
			}
			save_scope_as = GM_interacted_scope
		}
	}
	
	#return to GM tool
	
	option = {
		name = value_selection.1.a
		default_option = yes
		trigger_event = {
			id = gm_actions.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = value_selection.1.r
		default_option = yes
		remove_global_variable = GM_taking_action
		random_country = {
			limit = {
				has_modifier = GM_interacted_mod
			}
			remove_modifier = GM_interacted_mod
		}
	}
	

	#Reset
	
	option = {
		name = value_selection.1.b
		set_global_variable = {
			name = gm_variable_money
			value = 0
		}
		trigger_event = {
			id = value_selection.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = value_selection.1.d
		change_global_variable = {
			name = gm_variable_money
			add = 10000
		}
		trigger_event = {
			id = value_selection.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = value_selection.1.e
		change_global_variable = {
			name = gm_variable_money
			add = 100000
		}
		trigger_event = {
			id = value_selection.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = value_selection.1.f
		change_global_variable = {
			name = gm_variable_money
			add = 1000000
		}
		trigger_event = {
			id = value_selection.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = value_selection.1.h
		change_global_variable = {
			name = gm_variable_money
			subtract = 10000
		}
		trigger_event = {
			id = value_selection.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = value_selection.1.i
		change_global_variable = {
			name = gm_variable_money
			subtract = 100000
		}
		trigger_event = {
			id = value_selection.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = value_selection.1.j
		change_global_variable = {
			name = gm_variable_money
			subtract  = 1000000
		}
		trigger_event = {
			id = value_selection.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = value_selection.1.k
		change_global_variable = {
			name = gm_variable_money
			multiply = 10
		}
		trigger_event = {
			id = value_selection.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = value_selection.1.l
		change_global_variable = {
			name = gm_variable_money
			multiply = -1
		}
		trigger_event = {
			id = value_selection.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = value_selection.1.m
		scope:GM_interacted_scope = {
			add_treasury = global_var:gm_variable_money
		}
		remove_global_variable = GM_taking_action
		random_country = {
			limit = {
				has_modifier = GM_interacted_mod
			}
			remove_modifier = GM_interacted_mod
		}
	}
}
