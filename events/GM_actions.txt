namespace = gm_actions

gm_actions.1 = {
	type = country_event
	placement = ROOT
	title = gm_actions.1.t
	desc = gm_actions.1.u

	event_image = {
		texture = "gfx/event_pictures/void.dds"
	} 

	duration = 1

	immediate = {
		random_country = {
			limit = {
				has_modifier = GM_target
			}
			save_scope_as = scope_GM_target
		}
		random_country = {
			limit = {
				has_modifier = GM_interacted_mod
			}
			save_scope_as = GM_interacted_scope
		}
	}
	
	trigger = {
		#triggered by diplo action
	}
	
	#GM options
	
	option = {
		name = gm_actions.10.t
		trigger = {
			scope:GM_interacted_scope = {
				NOT = {
					has_modifier = GM_game_master
				}
				is_player = yes
			}
		}
		trigger_event = {
			id = gm_actions.10
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = gm_actions.1.a
		remove_global_variable = GM_taking_action
		scope:scope_GM_target = {
			remove_modifier = GM_target
		}
		scope:GM_interacted_scope = {
			remove_modifier = GM_interacted_mod
			add_modifier = {
				name = GM_target
			}
		}
	}
	
	option = {
		name = value_selection.1.t
		trigger_event = {
			id = value_selection.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = value_selection.2.t
		trigger_event = {
			id = value_selection.2
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = value_selection.pan.t
		trigger_event = {
			id = value_selection.pan
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = gm_actions.cancel
		default_option = yes
		remove_global_variable = GM_taking_action
		scope:GM_interacted_scope = {
			remove_modifier = GM_interacted_mod
		}
	}
	
}
