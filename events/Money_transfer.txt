money_transfer.2 = {
	type = country_event
	placement = ROOT
	title = money_transfer.2.t
	desc = money_transfer.2.u

	event_image = {
		texture = "gfx/event_pictures/void.dds"
	} 

	immediate = {
		save_scope_as = sender_scope
	}

	duration = 50
	
	trigger = {
		#triggered by diplo action
	}
	
	option = {
		name = gm_actions.cancel
		default_option = yes
		remove_variable = money_to_transfer_var
	}
	
	option = {
		name = money_transfer.2.a
		trigger_event = {
			id = money_transfer.1
			days = 0
			popup = yes
		}
	}
	
	option = {
		name = money_transfer.2.b
		set_global_variable = { 
			name = money_to_transfer_global_var_temp
			value = var:money_to_transfer_var
		}
		set_global_variable = { 
			name = money_to_transfer_global_var_temp_neg
			value = var:money_to_transfer_var
		}
		change_global_variable = {
			name = money_to_transfer_global_var_temp_neg
			multiply = -1
		}
		save_scope_as = sender_scope
		scope:target_country = {
			trigger_event = {
				id = money_transfer.3
				days = 0
				popup = yes
			}
		}
	}
}

money_transfer.3 = {
	type = country_event
	placement = ROOT
	title = money_transfer.3.t
	desc = money_transfer.3.u

	event_image = {
		texture = "gfx/event_pictures/void.dds"
	} 

	immediate = {
		set_variable = { 
			name = money_to_transfer_var_temp
			value = global_var:money_to_transfer_global_var_temp
		}
		set_variable = { 
			name = money_to_transfer_var_neg
			value = global_var:money_to_transfer_global_var_temp_neg
		}
	}

	duration = 50
	
	trigger = {
		#triggered by diplo action
	}
	
	option = {
		name = money_transfer.3.a
		default_option = yes
		scope:sender_scope = {
			remove_variable = money_to_transfer_var
		}
	}
	
	option = {
		name = money_transfer.3.b
		add_treasury = var:money_to_transfer_var_temp
		set_global_variable = { 
			name = money_to_transfer_global_var_temp_neg
			value = var:money_to_transfer_var_neg
		}
		scope:sender_scope = {
			add_treasury = global_var:money_to_transfer_global_var_temp_neg
			remove_variable = money_to_transfer_var
		}
	}
}
