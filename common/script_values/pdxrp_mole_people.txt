pdxrp_mole_people_personal_donation = {
    value = 0
    if = {
        limit = { has_variable = pdxrp_mole_people_donated }
        value = var:pdxrp_mole_people_donated
    }
}

# Country scope. India/China use the GP multiplier regardless of rank.
pdxrp_mole_people_rank_scale = {
    value = 1
    if = {
        limit = {
            OR = {
                country_rank = rank_value:great_power
                c:BIC ?= this
                c:HND ?= this
                c:CHI ?= this
            }
        }
        value = 0.2
    }
    else_if = {
        limit = { country_rank = rank_value:major_power }
        value = 0.5
    }
}

pdxrp_mole_people_participation_scale = {
    value = 1
    if = {
        limit = { pdxrp_mole_people_personal_donation <= 0 }
        value = 0.5
    }
}

# Apply rank and participation independently, so both reductions always stack.
pdxrp_mole_people_reward_scale = {
    value = global_var:pdxrp_mole_people_fund
    divide = 100000000
    max = 1
    multiply = pdxrp_mole_people_rank_scale
    multiply = pdxrp_mole_people_participation_scale
}
