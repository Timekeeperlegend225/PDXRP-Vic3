pdxrp_world_soul_temperature_text = {
    type = country
    text = {
        trigger = { pdxrp_is_campaign_host = yes }
        localization_key = pdxrp_world_soul_monaco_text
    }
    text = {
        trigger = {
            NOT = { pdxrp_is_campaign_host = yes }
            has_global_variable = pdxrp_world_soul_temperature
            global_var:pdxrp_world_soul_temperature < 80
        }
        localization_key = pdxrp_world_soul_below_80
    }
    text = {
        trigger = {
            NOT = { pdxrp_is_campaign_host = yes }
            has_global_variable = pdxrp_world_soul_temperature
            global_var:pdxrp_world_soul_temperature >= 80
            global_var:pdxrp_world_soul_temperature < 90
        }
        localization_key = pdxrp_world_soul_80_89
    }
    text = {
        trigger = {
            NOT = { pdxrp_is_campaign_host = yes }
            has_global_variable = pdxrp_world_soul_temperature
            global_var:pdxrp_world_soul_temperature >= 90
        }
        localization_key = pdxrp_world_soul_default_text
    }
}
