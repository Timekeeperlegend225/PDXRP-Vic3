AoE_stock_action = {
	name = "Take government ownership with stocks"
	desc = "Trade your leftover stock for nationalization"
	
	visible = { 
		var:Gilded_actions > 0
	}

	possible = {
		var:Gilded_actions > 0
	}

	effect = {
		set_variable = { 
			name = Gilded_actions_reward
			value = var:Gilded_actions
		}
		change_variable = { 
			name = Gilded_actions_reward
			multiply = 10000
		}
		add_treasury = var:Gilded_actions_reward
		add_modifier = { 
			name = HRP_stock_nationalization_modifier 
			multiplier = var:Gilded_actions
			years = 10
			decaying = yes
		}
		set_variable = { 
			name = Gilded_actions 
			value = 0
		}
	}
}

AoE_discourage_free_market = {
	name = "Discourage the free market"
	desc = "Promote nationaly owned building"
	
	visible = { 
		always = yes
	}

	possible = {
		NOT = {
			OR = {
				has_law = law_type:law_laissez_faire
				has_law = law_type:law_free_trade
				has_modifier = HRP_discourage_free_market_modifier
			}
		}
		authority >= 100
	}

	effect = {
		add_modifier = { 
			name = HRP_discourage_free_market_modifier 
			years = 1
		}
		custom_tooltip = {
			text = "Every building will get between 20% (fully self or state owned) and -10% (fully privately owned) thougput for 1 years depending of it's ownership"
			every_scope_building = {
				add_modifier = {
					name = HRP_discourage_free_marketbuilding_modifier
					multiplier = {
						add = private_ownership_fraction
						multiply = -3
						add = 2
					}
					months = 12
				}	
			}
		}
	}
}
