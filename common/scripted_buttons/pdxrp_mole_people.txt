pdxrp_mole_people_donate_one = {
    name = pdxrp_mole_people_donate_one
    desc = pdxrp_mole_people_donate_one_desc
    visible = { has_global_variable = pdxrp_mole_people_fund_open }
    possible = {
        has_global_variable = pdxrp_mole_people_fund_open
    }
    effect = {
        add_treasury = -1000000
        if = {
            limit = { NOT = { has_variable = pdxrp_mole_people_donated } }
            set_variable = { name = pdxrp_mole_people_donated value = 0 }
        }
        change_variable = { name = pdxrp_mole_people_donated add = 1000000 }
        change_global_variable = { name = pdxrp_mole_people_fund add = 1000000 }
    }
    ai_chance = { value = 0 }
}
