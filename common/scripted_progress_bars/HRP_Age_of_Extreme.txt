HRP_AoE_recovery = {
	name = "HRP_AoE_recovery_bar"
	desc = "HRP_AoE_recovery_desc"

	double_sided_bad = yes

	monthly_progress = {
		add = {
			desc = "Base Progress"
			value = global_var:recovery_flat_correction
		}
		add = {
			desc = "From state ownership"
			add = var:state_ownership_loc_var
			multiply = 4
		}
		if = {
			limit = {
				has_law = law_type:law_interventionism
			}
			add = {
				desc = "From Interventionism"
				value = 1
			}
		}
		else_if = {
			limit = {
				has_law = law_type:law_agrarianism
			}
			add = {
				desc = "From Agrarianism"
				value = 0.1
			}
		}
		else_if = {
			limit = {
				has_law = law_type:law_industry_banned
			}
			add = {
				desc = "From Industry Banned"
				value = 2
			}
		}
		else_if = {
			limit = {
				has_law = law_type:law_cooperative_ownership
			}
			add = {
				desc = "From Cooperative Ownership"
				value = 2
			}
		}
		else_if = {
			limit = {
				has_law = law_type:law_command_economy
			}
			add = {
				desc = "From Command Economy"
				value = 2
			}
		}
		if = {
			limit = {
				has_law = law_type:law_mercantilism
			}
			add = {
				desc = "From Mercantilism"
				value = 1
			}
		}
		else_if = {
			limit = {
				has_law = law_type:law_protectionism
			}
			add = {
				desc = "From Protectionism"
				value = 1
			}
		}
		else_if = {
			limit = {
				has_law = law_type:law_isolationism
			}
			add = {
				desc = "From Isolationism"
				value = 2
			}
		}
		if = {
			limit = {
				has_law = law_type:law_poor_laws
			}
			add = {
				desc = "From Poor Law"
				add = "institution_investment_level(institution_social_security)"
				multiply = 0.2
			}
		}
		else_if = {
			limit = {
				has_law = law_type:law_wage_subsidies
			}
			add = {
				desc = "From Wage Subsidies"
				add = "institution_investment_level(institution_social_security)"
				multiply = 0.3
			}
		}
		else_if = {
			limit = {
				has_law = law_type:law_old_age_pension
			}
			add = {
				add = "institution_investment_level(institution_social_security)"
				multiply = 0.4
			}
		}
		if = {
			limit = {
				has_variable = recoverysetup
			} 
			add = { 
				desc = "Initial recovery"
				name = HRP_AoE_recovery 
				add = var:starting_recovery
				add = global_var:GM_crash_countwodn
			}
		}
	}
	start_value = 0
	min_value = 0
	max_value = 1000
}

HRP_AoE_Left_E = {
	name = "HRP_AoE_Left_E_bar"
	desc = "HRP_AoE_Left_E_desc"

	default_bad = yes

	monthly_progress = {
		add = {
			desc = "Base Progress"
			value = global_var:extreme_flat_correction
		}
		add = {
			desc = "from Radicals"
			value = this.radical_population_fraction
			multiply = 5
		}
		subtract = {
			desc = "from Loyalists"
			value = this.loyalist_fraction
			multiply = 5
		}
		if = {
			limit = {
				any_political_Movement = {
					is_political_movement_type = movement_socialist
				}
			}
			add = {
				desc = "from Socialist Movement Activism"
				every_political_Movement = {
					limit = {
						is_political_movement_type = movement_socialist
					}
					add = this.political_movement_radicalism
					multiply = this.political_movement_support
					multiply = 5
				}
			}
		}
		if = {
			limit = {
				any_political_Movement = {
					is_political_movement_type = movement_communist
				}
			}
			add = {
				desc = "from Communist Movement Activism"
				every_political_Movement = {
					limit = {
						is_political_movement_type = movement_communist
					}
					add = this.political_movement_radicalism
					multiply = this.political_movement_support
					multiply = 5
				}
			}
		}
		if = {
			limit = {
				any_political_Movement = {
					is_political_movement_type = movement_anarchist

			}
			add = {
				desc = "from Anarchist Movement Activism"

					}
					add = this.political_movement_radicalism
					multiply = this.political_movement_support
					multiply = 5
				}
			}
		}
	}

	start_value = 0
	min_value = 0
	max_value = 1000

}
