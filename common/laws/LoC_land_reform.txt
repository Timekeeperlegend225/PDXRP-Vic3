REPLACE_OR_CREATE:law_rural_serfdom = {
	group = lawgroup_land_reform
	parent = law_serfdom # Inherits ideological stances from parent

	icon = "gfx/interface/icons/law_icons/serfdom.dds"

	progressiveness = -50

	is_visible = {
		OR = {
			is_in_geographic_region = geographic_region_india
			is_in_geographic_region = geographic_region_china
			THIS = c:NEP
		}
	}

	disallowing_laws = {
		law_interventionism
		law_laissez_faire
		law_command_economy
	}

	on_enact = {
		if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_commercialized_agriculture
					has_law_or_variant = law_type:law_homesteading
				}
			}
			custom_tooltip = farmers_pb_ig_shift_effect_negative_tt
		}
		if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_commercialized_agriculture
					has_law_or_variant = law_type:law_collectivized_agriculture
				}
			}
			custom_tooltip = rural_laborers_tu_ig_shift_effect_negative_tt
		}
	}

	on_activate = {
		if = {
			limit = {
				has_law_or_variant = law_type:law_rural_economy
			}
			active_law:lawgroup_economic_system ?= { 
				every_scope_amendment = {
					limit = {
						OR = {
							type = amendment_type:amendment_rural_tenant
							type = amendment_type:amendment_rural_comercialized
							type = amendment_type:amendment_rural_homesteadind
							type = amendment_type:amendment_rural_collectvized
						}
					}
					remove_amendment = yes
				}
			}
			active_law:lawgroup_economic_system ?= { 
				add_amendment = {
					type = amendment_rural_tenant
					sponsor = ROOT.ig:ig_landowners
					cooldown = 0
				}
			}
		}
	}

	modifier = {
		interest_group_ig_landowners_pol_str_mult = 0.25
		state_peasants_internal_migration_disallowed_bool = yes
		state_peasants_mass_migration_disallowed_bool = yes
		state_minimum_incorporated_subsistence_arable_land_add = 0.25
		building_minimum_incorporated_subsistence_employment_add = 0.10
	}

	# AI should never try to bring back serfdom
	ai_will_do = {
		always = no
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}
