# Refresh saved movements once per country; new movements use the updated definition.
on_monthly_pulse_country = {
    on_actions = { pdxrp_refresh_supremacist_ideologies }
}

pdxrp_refresh_supremacist_ideologies = {
    trigger = {
        NOT = { has_variable = pdxrp_supremacist_fascist_only_updated }
    }
    effect = {
        every_political_movement = {
            limit = { is_political_movement_type = movement_cultural_majority }
            set_core_ideology = ideology_fascist_movement
            add_character_ideology = ideology_fascist
            remove_character_ideology = ideology_ethno_nationalist
        }
        set_variable = pdxrp_supremacist_fascist_only_updated
    }
}
