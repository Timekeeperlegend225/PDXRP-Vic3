on_law_activated = {
    effect = {
        if = {
            limit = {
                has_variable = save_edit_secret_society_active
                is_revolutionary = no
                exists = ig:ig_armed_forces
                active_law:lawgroup_governance_principles = {
                    NOT = { has_amendment = amendment_type:save_edit_secret_society }
                }
            }
            ig:ig_armed_forces = { save_scope_as = save_edit_secret_society_sponsor }
            active_law:lawgroup_governance_principles = {
                add_amendment = {
                    type = save_edit_secret_society
                    sponsor = scope:save_edit_secret_society_sponsor
                    cooldown = 0
                    timeout = 0
                }
            }
        }
    }
}

on_revolution_start = {
    effect = {
        scope:target ?= {
            remove_variable = save_edit_secret_society_active
            every_law = {
                every_scope_amendment = {
                    limit = { type = amendment_type:save_edit_secret_society }
                    remove_amendment = yes
                }
            }
        }
    }
}
