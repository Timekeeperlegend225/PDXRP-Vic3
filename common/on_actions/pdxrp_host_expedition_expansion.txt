on_monthly_pulse_country = {
    on_actions = { pdxrp_host_expedition_maintenance }
}
pdxrp_host_expedition_maintenance = {
    effect = {
        # Reattach the updated shared megaboost once for existing ZIP/Monaco saves.
        if = {
            limit = {
                OR = { c:ZIP ?= THIS c:MOA ?= THIS }
                has_modifier = pdxrp_zip_bomb_collective
                NOT = { has_variable = pdxrp_expedition_ship_efficiency_600_refreshed }
            }
            remove_modifier = pdxrp_zip_bomb_collective
            add_modifier = { name = pdxrp_zip_bomb_collective }
            set_variable = pdxrp_expedition_ship_efficiency_600_refreshed
        }
        if = {
            limit = { c:ZIP ?= THIS }
            if = {
                limit = { NOT = { has_modifier = pdxrp_expedition_birth_penalty } }
                add_modifier = { name = pdxrp_expedition_birth_penalty }
            }
            if = {
                limit = {
                    has_modifier = pdxrp_zip_bomb_collective
                    NOT = { has_variable = pdxrp_expedition_combat_100_refreshed }
                }
                remove_modifier = pdxrp_zip_bomb_collective
                add_modifier = { name = pdxrp_zip_bomb_collective }
                set_variable = pdxrp_expedition_combat_100_refreshed
            }
        }
        if = {
            limit = {
                any_scope_pop = { culture = cu:baganda }
                NOT = { any_political_movement = { is_political_movement_type = movement_pdxrp_pop_minority } }
                NAND = {
                    has_law_or_variant = law_type:law_multicultural
                    has_law_or_variant = law_type:law_anarchy
                }
            }
            create_political_movement = { type = movement_pdxrp_pop_minority }
        }
    }
}
