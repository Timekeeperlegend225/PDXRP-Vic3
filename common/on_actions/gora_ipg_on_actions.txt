# GORA UI+ — Improved Parliament Graph: on-action hooks
#
# All recalcs are gated by `is_player = yes` so work is limited to the
# human player's country. This is what keeps the mod safe for multiplayer:
# each client runs the same calc on its own scope, producing deterministic
# variables that survive a save/load identically across clients.

# Initial seed: run once on save load so the panel is populated before the
# player even opens the Government screen.
on_game_started_after_lobby = {
	on_actions = {
		delay = { days = 1 }
		ipg_on_game_started
	}
}

# Slow heartbeat: monthly is plenty for IG clout drift and avoids piling up
# work on the daily tick (important for large MP games).
on_monthly_pulse_country = {
	on_actions = { ipg_on_player_recalc }
}

# Snap update right after an election concludes — seat counts can swing hard.
on_election_campaign_end = {
	on_actions = { ipg_on_player_recalc }
}

# Laws can flip enfranchisement (and therefore the Electoral view's existence).
on_law_activated = {
	on_actions = { ipg_on_law_activated }
}

# Election months also tick a separate pulse — reuse the same recalc.
on_monthly_pulse_country_elections = {
	on_actions = { ipg_on_player_recalc }
}

# ─── Bridges (scope guard + dispatch) ─────────────────────────────────────

ipg_on_game_started = {
	effect = {
		every_country = {
			limit = { 
				AND = {
					is_player = yes 
					NOT = { has_variable = bpmparl_last_refresh_day }
				}
			}
			ipg_recalculate_parliament = yes
		}
	}
}

ipg_on_player_recalc = {
	effect = {
		if = {
			limit = { 
				AND = {
					is_player = yes 
					NOT = { has_variable = bpmparl_last_refresh_day }
				}
			}
			ipg_recalculate_parliament = yes
		}
	}
}

# on_law_activated fires in Law scope; hop to the owner country first.
ipg_on_law_activated = {
	effect = {
		owner = {
			if = {
				limit = { 
					AND = {
						is_player = yes 
						NOT = { has_variable = bpmparl_last_refresh_day }
					}
				}
				ipg_recalculate_parliament = yes
			}
		}
	}
}
