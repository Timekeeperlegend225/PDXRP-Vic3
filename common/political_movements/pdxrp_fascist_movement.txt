REPLACE:movement_fascist = {
	category = movement_category_ideological

	active_law_radicalism_multiplier = 1

	ideology = ideology_fascist_movement
	character_ideologies = {
		ideology_fascist
		ideology_integralist
		ideology_corporatist_leader
	}

	creation_trigger = {
		has_technology_researched = political_agitation
		country_should_exclusively_focus_on_modernization = no
	}
	creation_weight = {
		value = 100
		if = {
			limit = {
				owner ?= { # If a movement of this type exists in a neighbouring country, make it more likely to spawn here as well
					any_neighbouring_state = {
						owner = {
							prestige > ROOT.owner.prestige
							any_political_movement = {
								is_political_movement_type = movement_fascist
							}
						}
					}
				}
			}
			multiply = {
				value = 2
			}
		}
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
		if = {
			limit = {
				owner ?= {
					any_primary_culture = {
						has_discrimination_trait = language_lusophone
					}
				}
			}
			set_core_ideology = ideology_integralist_movement
		}
	}

	character_support_trigger = {
        NOT = { has_template = pdxrp_donald_macdonald }
		OR = {
			has_ideology = ideology:ideology_fascist
			has_ideology = ideology:ideology_integralist
			has_ideology = ideology:ideology_corporatist_leader
			has_ideology = ideology:ideology_ethno_nationalist
		}
	}
	character_support_weight = {
		value = 200
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_integralist
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_corporatist_leader
					has_ideology = ideology:ideology_ethno_nationalist
				}
			}
			multiply = {
				value = 2
			}
		}
	}

	pop_support_trigger = {
		NOT = {
			is_pop_type = slaves
		}
	}

	pop_support_factors = {
		movement_support_high_literacy
		movement_support_middle_below_expected_sol
		movement_support_radicalization
		movement_support_unemployed
		movement_support_capitalists
		movement_support_aristocrats
		movement_support_shopkeepers
		movement_support_officers
		movement_support_farmers
		movement_support_clerks
		movement_support_soldiers
		movement_support_number_of_claims
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
						OR = {
							is_being_suppressed = yes
							is_being_bolstered = yes
						}
					}
				}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}

		if = {
			limit = {
				strata = upper
			}
			if = {
				limit = {
					is_pop_type = capitalists
				}
				add = {
					value = 12
					desc = "POP_CAPITALISTS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = aristocrats
				}
				add = {
					value = 5
					desc = "POP_ARISTOCRATS" # Fascism is too populist
				}
			}
			else = {
				add = {
					value = 5
					desc = "UPPER_NO_ICON"
				}
			}

			if = {
				limit = {
					owner ?= {
						any_political_movement = {
							OR = {
								is_political_movement_type = movement_socialist
								is_political_movement_type = movement_anarchist
								is_political_movement_type = movement_communist
								is_political_movement_type = movement_labor
								is_political_movement_type = movement_nihilist
							}
							political_movement_support >= 0.05
						}
					}
				}
				multiply = {
					desc = "STRONG_SOCIALIST_MOVEMENT"
					value = 1.5
					owner ?= {
						every_political_movement = {
							limit = {
								OR = {
									is_political_movement_type = movement_socialist
									is_political_movement_type = movement_anarchist
									is_political_movement_type = movement_communist
									is_political_movement_type = movement_labor
									is_political_movement_type = movement_nihilist
								}
							}
							add = this.political_movement_radicalism
						}
					}
				}
			}
			if = {
				limit = {
					owner = {
						number_of_claims > 0
					}
				}

				add = {
					value = culture.culture_current_fervor # Add a base value for having claims at all
					divide = 100 # Culture is 1-100
					multiply = {
						value = owner.number_of_claims
					}
					max = 10
					desc = "NUMBER_OF_CLAIMS"
				}
			}
		}
		else_if = {
			limit = {
				strata = middle
			}
			if = {
				limit = {
					is_pop_type = shopkeepers
				}
				add = {
					value = 15
					desc = "POP_SHOPKEEPERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = bureaucrats
				}
				add = {
					value = 8
					desc = "POP_BUREAUCRATS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = officers
				}
				add = {
					value = 8
					desc = "POP_OFFICERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = farmers
				}
				add = {
					value = 5
					desc = "POP_FARMERS"
				}
			}
			else = {
				add = {
					value = 4
					desc = "MIDDLE_NO_ICON"
				}
			}

			if = {
				limit = {
					owner ?= {
						any_political_movement = {
							OR = {
								is_political_movement_type = movement_socialist
								is_political_movement_type = movement_anarchist
								is_political_movement_type = movement_communist
								is_political_movement_type = movement_labor
								is_political_movement_type = movement_nihilist
							}
							political_movement_support >= 0.05
						}
					}
				}
				multiply = {
					desc = "STRONG_SOCIALIST_MOVEMENT"
					value = 1
					owner ?= {
						every_political_movement = {
							limit = {
								OR = {
									is_political_movement_type = movement_socialist
									is_political_movement_type = movement_anarchist
									is_political_movement_type = movement_communist
									is_political_movement_type = movement_labor
									is_political_movement_type = movement_nihilist
								}
							}
							add = this.political_movement_radicalism
						}
					}
				}
			}
			if = {
				limit = {
					owner = {
						number_of_claims > 0
					}
				}
				add = {
					value = culture.culture_current_fervor # Add a base value for having claims at all
					divide = 100 # Culture is 1-100
					multiply = {
						value = owner.number_of_claims
					}
					max = 10
					desc = "NUMBER_OF_CLAIMS"
				}
			}
		}
		else_if = {
			limit = {
				strata = lower
			}
			if = {
				limit = {
					is_pop_type = clerks
				}
				add = {
					value = 5
					desc = "POP_CLERKS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = soldiers
				}
				add = {
					value = 3
					desc = "POP_SOLDIERS"
				}
			}
			else = {
				add = {
					value = 2
					desc = "LOWER_NO_ICON"
				}
			}
			if = {
				limit = {
					OR = {
						is_pop_type = clerks
						is_pop_type = soldiers
					}
					owner = {
						number_of_claims > 0
					}
				}
				add = {
					value = culture.culture_current_fervor # Add a base value for having claims at all
					divide = 125 # Culture is 1-100
					multiply = {
						value = owner.number_of_claims
					}
					max = 5
					desc = "NUMBER_OF_CLAIMS"
				}
			}
		}

		if = {
			limit = {
				standard_of_living < state.average_expected_sol
				OR = {
					strata = middle
					strata = upper
				}
			}
			add = { # Increases likelihood of support if impoverished
				desc = "IS_BELOW_EXPECTED_SOL"
				value = state.average_expected_sol
				subtract = standard_of_living
				max = 10
			}
		}
		else_if = {
			limit = {
				standard_of_living > state.average_expected_sol
				OR = {
					strata = middle
					strata = upper
				}
			}
			add = { # Decreases likelihood of support if well-off
				desc = "IS_ABOVE_EXPECTED_SOL"
				value = state.average_expected_sol
				subtract = standard_of_living
				min = -5
			}
		}

		if = {
			limit = {
				is_employed = no
			}
			multiply = {
				value = 1.5
				desc = "UNEMPLOYED"
			}
		}

		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.10
		}

		multiply = {
			value = 0.011
			multiply = pop_acceptance
			desc = "ACCEPTANCE"
		}

		if = {
			limit = {
				OR = {
					strata = upper
					strata = middle
					is_pop_type = clerks
				}
			}
			multiply = { # Multiplied by 2x max
				desc = "POP_RADICALS"
				value = pop_radical_fraction
				multiply = 2
				add = 1
			}
		}
		else = {
			multiply = { # Multiplied by 1.5x max
				desc = "POP_RADICALS"
				value = pop_radical_fraction
				divide = 2
				add = 1
			}
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = 0.25
			add = scope:clout
			add = { value = this.turmoil divide = 2 }
			subtract = this.loyalists
		}
	}

	secession = {
		possible = {
			political_movement_support >= 0.05
			scope:clout <= 0
		}

		weight = {
			value = 1000
		}
	}

	additional_radicalism_factors = {
		owner = {
			if = {
				limit = {
					owner.number_of_claims > 0
				}

				add = {
					value = 5 # Add a base value for having claims at all
					add = owner.number_of_claims
					multiply = 0.025
					max = 0.3
					desc = "NUMBER_OF_CLAIMS"
				}
			}
		}

		if = {
			limit = {
				owner = {
					any_neighbouring_state = {
						owner = {
							NOT = { THIS = ROOT.owner }
							any_political_movement = {
								is_political_movement_type = movement_fascist
								is_insurrectionary = yes
							}
						}
					}
				}
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}

		if = { # Support translates 1:1 to Activism
			limit = {
				owner = {
					NAND = {
						has_law = law_type:law_single_party_state
						ruler.interest_group.party ?= {
							is_party_type = fascist_party
						}
					}
				}
			}
			add = {
				desc = "FASCIST_MOVEMENT_WILL_TO_POWER"
				add = this.political_movement_support
				max = 0.5
			}
		}
		if = {
			limit = {
				this.days_since_movement_defeated <= very_long_modifier_time
			}
			add = { # (x-y)/y - ([1, 7300]-7300)/7300
				desc = "movement_suppressed_after_civil_war"
				value = this.days_since_movement_defeated
				subtract = very_long_modifier_time
				divide = {
					value = very_long_modifier_time
				}
			}
		}
	}
}
