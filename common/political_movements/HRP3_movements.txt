movement_communalist = {
	category = movement_category_ideological
	
	ideology = ideology_communalism
	character_ideologies = {
		ideology_communalism
	}
	
	creation_trigger = {
		has_global_variable = age_of_extreme_gm
	}	
	creation_weight = {
		value = 100
	}	
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}
	disband_trigger = {
		has_modifier = remove_socialism
	}
	
	character_support_trigger = {
		OR = {
			has_ideology = ideology:ideology_communalism
			is_interest_group_type = ig_trade_unions
			is_interest_group_type = ig_devout
		}
	}
	character_support_weight = {
		value = 50
		if = {
			limit = {
				has_ideology = ideology:ideology_communalism
			}
			multiply = {
				value = 40
			}
		}
		if = {
			limit = {
				is_interest_group_type = ig_devout
				has_ideology = ideology:ideology_social_democrat
			}
			multiply = {
				value = 4
			}
		}
	}

	pop_support_trigger = {
		NOR = {
			is_pop_type = capitalists
			is_pop_type = aristocrats
			is_pop_type = slaves
		}
		OR = {
			religion = {
				shares_heritage_trait_with_state_religion = prev.owner #Only people of the same religious group will support
			}
			owner = {
				has_law = law_type:law_state_atheism 
			}
		}
	}

	pop_support_factors = {
		movement_support_low_urbanization
		movement_support_high_literacy
		movement_support_below_expected_sol
		movement_support_radicalization
		movement_support_academics
		movement_support_laborers
		movement_support_machinists
		movement_support_peasants
		movement_support_rural
		movement_support_clergymen
	}
	
	pop_support_weight = {	
		add = {
			desc = "URBAN_STATE" # Increases support in rural states, decreases in urban
			value = 0.25
			subtract = state.state_urbanization_rate
			multiply = 5
		}
		if = {
			limit = { 
				owner = {
					has_law = law_type:law_charitable_health_system 
				}
			}
			add = {
				value = "owner.institution_investment_level(institution_health_system)"
				multiply = 0.6
				desc = "INSTITUTION_FUNDING_LEVEL_health_system"
			}
		}
		if = {
			limit = { 
				owner = {
					any_interest_group = {
						is_powerful = yes
						leader = {
							has_ideology = ideology:ideology_communalism
						}
					} 
				}
			}
			add = {
				value = 5
				desc = "COMMUNALIST_IG"
			}
		}
	
		if = {
			limit = {
				strata = middle
			}
			if = {
				limit = {
					is_pop_type = clergymen
				}
				add = {
					value = 20
					desc = "POP_CLERGYMEN"
				}
			}
			else_if = {
				limit = {
					is_pop_type = academics
				}
				add = {
					value = 3
					desc = "POP_ACADEMICS"
				}
			}
		}
		else_if = {
			limit = {
				strata = lower
			}
			if = {
				limit = {
					is_pop_type = laborers
				}
				add = {
					value = 15
					desc = "POP_LABORERS"
				}
			}		
			else_if = {
				limit = {
					is_pop_type = machinists
				}
				add = {
					value = 20
					desc = "POP_MACHINISTS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = clerks
				}
				add = {
					value = 6
					desc = "POP_CLERKS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = peasants 
				}
				add = {
					value = 10
					desc = "POP_PEASANTS"
				}
			}
			else = {
				add = {
					value = 6
					desc = "LOWER_NO_ICON"
				}
			}
		}
		if = {
			limit = { 
				owner = {
					has_law = law_type:law_religious_schools 
				}
			}
			multiply = {
				value = "owner.institution_investment_level(institution_schools)"
				multiply = 0.05
				add = 1
				desc = "INSTITUTION_FUNDING_LEVEL_schools"
			}
		}
		if = {
			limit = { 
				owner = {
					ig:ig_devout = {
						is_powerful = yes
					} 
				}
			}
			multiply = {
				value = 1.2
				desc = "RELIGIOUS_SOCIETY"
			}
		}
		if = {
			limit = { 
				owner = {
					ig:ig_trade_unions = {
						is_powerful = yes
						leader = {
							has_ideology = ideology:ideology_communalism
						}
					} 
				}
			}
			multiply = {
				value = 1.2
				desc = "RELIGIOUS_UNION"
			}
		}
		if = {
			limit = { 
				owner = {
					has_law = law_type:law_charitable_health_system 
				}
			}
			multiply = {
				value = "owner.institution_investment_level(institution_health_system)"
				multiply = 0.1
				add = 1
				desc = "INSTITUTION_FUNDING_LEVEL_health_system"
			}
		}
		if = {
			limit = {
				standard_of_living < state.average_expected_sol
				strata = lower
			}
			add = { # Increases likelihood of support if impoverished
				desc = "IS_BELOW_EXPECTED_SOL"
				value = state.average_expected_sol
				subtract = standard_of_living
				max = 10
			}
		}
		else_if = {
			limit = {
				standard_of_living > state.average_expected_sol
				strata = lower
			}
			add = { # Decreases likelihood of support if well-off
				desc = "IS_ABOVE_EXPECTED_SOL"
				value = state.average_expected_sol
				subtract = standard_of_living
				multiply = 0.5
			}
		}

		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.20
		}

		multiply = { # Multiplied by 1.5x max
			desc = "POP_RADICALS"
			value = pop_radical_fraction
			multiply = 0.5
			add = 1
		}
	}
	
	revolution = {
		possible = {
			political_movement_support >= 0.05
		}
		
		weight = {
			value = 100
		}	
	}
	
	secession = {
		possible = {
			political_movement_support >= 0.05
			scope:clout <= 0
		}
		
		weight = {
			value = 1000
		}		
	}	
	
	additional_radicalism_factors = {
		owner = {
			if = {
				limit = { has_institution = institution_workplace_safety }
				
				add = {
					value = 3
					subtract = "institution_investment_level(institution_workplace_safety)"
					multiply = 0.05
					desc = "INSTITUTION_FUNDING_LEVEL_workplace_safety"
				}
			}
			else = {
				add = {
					value = 0.1
					desc = "NO_INSTITUTION_workplace_safety"
				}		
			}
			if = {
				limit = { has_law = law_type:law_state_atheism }
				add = {
					value = 0.5	
					desc = "STATE_ATHEISM"
				}
			}
		}
	}
}

modern_elitist_movement = {
	category = movement_category_ideological
	
	ideology = ideology_modern_elitist_movement
	character_ideologies = {
		ideology_protectionist
		ideology_market_liberal
		ideology_corporatist_leader
	}
	
	creation_trigger = {
		has_technology_researched = academia
		OR = {
			AND = {
				OR = {
					country_is_in_south_america = yes
					country_is_in_central_america = yes
				}
				country_rank <= rank_value:major_power
			}
			is_country_type = colonial
			is_country_type = unrecognized
		}
		NAND = {
			has_law = law_type:law_protectionism
			OR = {
				has_law = law_type:law_interventionism
				has_law = law_type:law_command_economy
				has_law = law_type:law_laissez_faire
			}
			has_law = law_type:law_compulsory_primary_school
		}
		game_date < 1887.01.01
	}	
	creation_weight = {
		value = 100
	}	
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}
	disband_trigger = {
		has_modifier = remove_elistism
	}
	
	character_support_trigger = {
		trigger_if = {
			limit = {
				is_in_exile_pool = yes
			}
			OR = {
				has_ideology = ideology:ideology_reformer
				has_ideology = ideology:ideology_protectionist

				has_ideology = ideology:ideology_abolitionist

				has_ideology = ideology:ideology_reformer

				has_ideology = ideology:ideology_populism

					OR = {
						has_ideology = ideology:ideology_modernizer
						has_ideology = ideology:ideology_liberal
						has_ideology = ideology:ideology_liberal_modern
						has_ideology = ideology:ideology_plutocratic
					}
				}
			}
		}
	}

				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_positivist
					has_ideology = ideology:ideology_abolitionist
				}
			}
			multiply = {
				value = 3
			}
		}
		if = {
			limit = {
				interest_group ?= {
					OR = {
						has_ideology = ideology:ideology_modernizer
						has_ideology = ideology:ideology_liberal
						has_ideology = ideology:ideology_liberal_modern
						has_ideology = ideology:ideology_plutocratic

			}

				value = 2
