# Supremacist character ideology options and explicit upper-strata attraction bonuses.
REPLACE:movement_cultural_majority = {
	category = movement_category_cultural_majority

	ideology = ideology_fascist_movement
	character_ideologies = {
		ideology_fascist
	}

	can_cause_obstinance = yes

	creation_trigger = {
		NOR = { # Will not spawn if you're an un-unified German or Italian state
			AND = {
				OR = {
					country_has_primary_culture = cu:north_german
					country_has_primary_culture = cu:south_german
				}
				NOT = {
					c:GER ?= this
				}
			}
			AND = {
				OR = {
					country_has_primary_culture = cu:north_italian
					country_has_primary_culture = cu:south_italian
				}
				NOR = {
					c:ITA ?= this
					c:AUS ?= this
					c:KUK ?= this
				}
			}
		}
	}

	creation_weight = {
		value = 100
	}
	on_created = {}

	character_support_trigger = {
		culture = {
			is_primary_culture_of = root.owner
		}
		OR = {
			has_ideology = ideology:ideology_fascist
			has_ideology = ideology:ideology_jingoist_leader
			has_ideology = ideology:ideology_ethno_nationalist
		}
		NOT = {
			has_variable = anti_nationalist_var
		}
	}
	character_support_weight = {
		value = 100
        # Donald MacDonald specifically campaigns for the national Supremacist movement.
        if = {
            limit = { has_template = pdxrp_donald_macdonald }
            add = 1000000
        }

		if = {
			limit = {
				has_ideology = ideology:ideology_ethno_nationalist
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_jingoist_leader
				}
			}
			multiply = {
				value = 2
			}
		}
		else_if = {
			limit = {
				OR = {
					has_progressive_ideology = yes
					has_socialist_ideology = yes
				}
			}
			multiply = 0.05
		}
	}

	pop_support_trigger = {
		culture = {
			is_primary_culture_of = root.owner
		}
	}

	pop_support_factors = {
		movement_support_capitalists
		movement_support_aristocrats
		movement_support_national_awakening
		movement_support_cultural_fervor
		movement_support_shopkeepers
		movement_support_officers
		movement_support_farmers
		movement_support_clerks
		movement_support_soldiers
		movement_support_peasants
		movement_support_colonial_settlers
		movement_support_nationalism_tech
		movement_support_pan_nationalism_tech
	}

	pop_support_weight = {
		# Additional profession bonuses matching Fascist movement values.
		if = {
			limit = { is_pop_type = capitalists }
			add = { value = 12 desc = "POP_CAPITALISTS" }
		}
		else_if = {
			limit = { is_pop_type = aristocrats }
			add = { value = 5 desc = "POP_ARISTOCRATS" }
		}

		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
					OR = {
						is_being_suppressed = yes
						is_being_bolstered = yes
					}
				}
			}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}

		if = {
			limit = {
				strata = upper
			}
			add = {
				value = 10
				desc = "UPPER_NO_ICON"
			}
		}
		else_if = {
			limit = {
				strata = middle
			}
			if = {
				limit = {
					is_pop_type = shopkeepers
				}
				add = {
					value = 25
					desc = "POP_SHOPKEEPERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = officers
				}
				add = {
					value = 20
					desc = "POP_OFFICERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = farmers
				}
				add = {
					value = 15
					desc = "POP_FARMERS"
				}
			}
			else = {
				add = {
					value = 12
					desc = "MIDDLE_NO_ICON"
				}
			}
		}
		else_if = {
			limit = {
				strata = lower
			}
			if = {
				limit = {
					is_pop_type = clerks
				}
				add = {
					value = 25
					desc = "POP_CLERKS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = soldiers
				}
				add = {
					value = 20
					desc = "POP_SOLDIERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = peasants
				}
				add = {
					value = 10
					desc = "POP_PEASANTS"
				}
			}
			else = {
				add = {
					value = 5
					desc = "LOWER_NO_ICON"
				}
			}
		}

		if = {
			limit = {
				state = {
					is_incorporated = no
				}
			}
			multiply = {
				value = 1.5
				desc = "COLONIAL_SETTLERS"
			}
		}

		if = {
			limit = {
				owner ?= {
					has_technology_researched = pan-nationalism
				}
			}
			multiply = {
				value = 1.5
				desc = "PAN_NATIONALISM_TECH"
			}
		}
		else_if = {
			limit = {
				owner ?= {
					has_technology_researched = nationalism
				}
			}
			multiply = {
				value = 1.25
				desc = "NATIONALISM_TECH"
			}
		}

		multiply = { # Range from x0.775 to 2.275
			desc = "POP_CULTURE_FERVOR"
			value = culture.culture_current_fervor
			subtract = 15
			multiply = 0.015
			add = 1
		}

		if = {
			limit = {
				culture ?= {
					culture_has_national_awakening = yes
				}
			}
			multiply = {
				value = 1.5
				desc = "POP_ACTIVE_NATIONAL_AWAKENING"
			}
		}

		if = { # If a more radical movement exists, it'll poach radicals from this one.
			limit = {
				owner ?= {
					any_political_movement = {
						OR = {
							is_political_movement_type = movement_reactionary
							is_political_movement_type = movement_fascist
						}
					}
				}
			}
			multiply = {
				desc = "POP_RADICAL_MOVEMENT_LOSSES"
				value = 1
				subtract = pop_radical_fraction
			}
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = 1

			if = {
				limit = {
					is_capital = no
					is_homeland_of_country_cultures = owner
					primary_cultures_percent_state = {
						target_country = owner
						value >= 0.05
					}
				}
				add = 10

				if = {
					limit = {
						primary_cultures_percent_state = {
							target_country = owner
							value >= 0.75
						}
					}
					add = 1000
				}
				else_if = {
					limit = {
						primary_cultures_percent_state = {
							target_country = owner
							value >= 0.5
						}
					}
					add = 500
				}
				else_if = {
					limit = {
						primary_cultures_percent_state = {
							target_country = owner
							value >= 0.25
						}
					}
					add = 100
				}
				else_if = {
					limit = {
						primary_cultures_percent_state = {
							target_country = owner
							value >= 0.1
						}
					}
					add = 50
				}
			}
		}
	}

	secession = {
		possible = {
			political_movement_support >= 0.05
			scope:clout <= 0
		}

		weight = {
			value = 1000
		}

		state_weight = {
			value = 0

			if = {
				limit = {
					is_capital = no
					is_homeland_of_country_cultures = owner
					primary_cultures_percent_state = {
						target_country = owner
						value >= 0.05
					}
				}
				add = { # These three values must add up to more than 1 - 5 x Obstinance, 1/2 Turmoil, percent of pops of primary cultures
					value = -1
					add = "this.primary_cultures_percent_state(owner)"
					add = {
						value = this.turmoil
						multiply = 0.5
					}
					add = {
						value = this.obstinance
						multiply = 5
				}
					multiply = 100
				}
			}
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = {
				owner.ruler ?= {
					NOT = { culture = { is_primary_culture_of = root.owner } }
				}
			}

			add = {
				value = 0.1
				desc = "RULER_NOT_PRIMARY_CULTURE"
			}
		}

		add = {
			value = "owner.average_sol"
			subtract = "owner.average_sol_for_primary_cultures"
			multiply = 0.05
			max = 0.5
			min = 0
			desc = "STANDARD_OF_LIVING_FOR_PRIMARY_CULTURES"
		}

		if = {
			limit = {
				owner ?= {
					has_technology_researched = nationalism
					any_scope_homeland_state = {
						NOT = {
							this.owner ?= {
								any_primary_culture = {
									is_primary_culture_of = root.owner
								}
							}
						}
					}
				}
			}
			add = {
				owner ?= {
					every_scope_homeland_state = {
						limit = {
							NOT = {
								this.owner ?= {
									any_primary_culture = {
										is_primary_culture_of = root.owner
									}
								}
							}
						}
						add = 0.025
					}
				}
				max = 0.25
				desc = "MOVEMENT_RADICALISM_UNOWNED_HOMELAND_STATES_FACTOR"
			}
		}

		if = {
			limit = {
				owner = {
					any_neighbouring_state = {
						owner = {
							NOT = { THIS = ROOT.owner }
							any_political_movement = {
								OR = {
									AND = {
										is_political_movement_type = movement_cultural_majority
										prev = {
											any_primary_culture = {
												is_primary_culture_of = root.owner
											}
										}
									}
									AND = {
										is_political_movement_type = movement_cultural_minority
										culture ?= { is_primary_culture_of = root.owner }
									}
								}
								is_insurrectionary = yes
							}
						}
					}
				}
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_CULTURE_REVOLUTION"
				value = 0.15
			}
		}

		if = {
			limit = {
				this.days_since_movement_defeated <= very_long_modifier_time
			}
			add = { # (x-y)/y - ([1, 7300]-7300)/7300
				desc = "movement_suppressed_after_civil_war"
				value = this.days_since_movement_defeated
				subtract = very_long_modifier_time
				divide = {
					value = very_long_modifier_time
				}
			}
		}
	}
}
