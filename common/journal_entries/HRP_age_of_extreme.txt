je_HRP_recovery_and_extremism = {
	icon = "gfx/interface/icons/generic_icons/company_prosperity.dds"

	group = je_group_historical_content

	scripted_progress_bar = HRP_AoE_recovery
	scripted_progress_bar = HRP_AoE_Left_E
	scripted_progress_bar = HRP_AoE_Right_E

	immediate = {
		if = {
			limit = {
				has_variable = national_integration_var
			}
			set_variable = {
				name = starting_recovery
				value = 1
			}
			change_variable = {
				name = starting_recovery
				subtract = var:national_integration_var
			}
			change_variable = {
				name = starting_recovery
				multiply = 1000
			}
		}
		else = {
			set_variable = {
				name = starting_recovery
				value = global_var:GM_crash_countwodn
			}
			change_variable = {
				name = starting_recovery
				multiply = 2
			}
		}
		set_variable = recoverysetup 
		remove_modifier = HRP_crash
		scope:journal_entry = { 
			remove_modifier = HRP_crash
			add_modifier = {
				name = HRP_crash
				multiplier = {
					add = 900
					subtract = var:starting_recovery
					divide  = 1000
				}
			}
		}
		trigger_event = {
			id = hrp3_extremes.1
			days = 0
			popup = yes
		}
		trigger_event = {
			id = hrp3_extremes.2
			days = 0
			popup = yes
		}
	}

	complete = {
		custom_tooltip = {
			text = "Our recovery is complete"
			scope:journal_entry = {
				"scripted_bar_progress(HRP_AoE_recovery)" >= 1000
			}
		}
	}
	
    invalid = {
		custom_tooltip = {
			text = "Age of Extreme is finished"
			NOT = {
				has_global_variable = age_of_extreme_gm
			}
		}
    }
	
	scripted_button = AoE_stock_action
	scripted_button = AoE_pause_interest
	scripted_button = AoE_inflate_currency
	scripted_button = AoE_discourage_free_market
	scripted_button = AoE_socialized_cooperation
	scripted_button = AoE_national_companies
	scripted_button = AoE_emergenci_nationalization
	scripted_button = AoE_scapgoat_elite
	scripted_button = AoE_scapgoat_minorities
	scripted_button = AoE_scapgoat_other_country
	scripted_button = AoE_seize_capitalist_wealth
	scripted_button = AoE_deploy_army
	scripted_button = AoE_autarky
	scripted_button = AoE_suppress_socialists
	scripted_button = AoE_suppress_nationalist

	status_desc = {
	}

	on_monthly_pulse = {
		effect = {
			remove_variable = recoverysetup
			HRP_AoE_recovery_progress = yes 
			HRP_AoE_Left_E_progress = yes
			HRP_AoE_Right_E_progress = yes
			set_variable = { 
				name = recovery_var 
				value = "scope:journal_entry.scripted_bar_progress(HRP_AoE_recovery)" 
			}
			remove_modifier = HRP_stability_visual_modifier
			add_modifier = {
				name = HRP_stability_visual_modifier
				multiplier = var:tempnationalstab
			}
			set_variable = { 
				name = left_extreme_var
				value = "scope:journal_entry.scripted_bar_progress(HRP_AoE_Left_E)" 
			}
			set_variable = { 
				name = right_extreme_var 
				value = "scope:journal_entry.scripted_bar_progress(HRP_AoE_Right_E)" 
			}
			set_variable = { 
				name = both_extreme_var
				value = var:left_extreme_var
			}
			change_variable = { 
				name = both_extreme_var
				add = var:right_extreme_var
			}
			scope:journal_entry = { 
				remove_modifier = HRP_crash
				remove_modifier = HRP_left_extreme_modifier
				remove_modifier = HRP_right_extreme_modifier
				add_modifier = {
					name = HRP_crash
					multiplier = {
						add = 1000
						subtract = var:recovery_var
						divide  = 1000
					}
				}
				add_modifier = {
					name = HRP_left_extreme_modifier
					multiplier = {
						add = var:left_extreme_var
						divide  = 1000
					}
				}
				add_modifier = {
					name = HRP_right_extreme_modifier
					multiplier = {
						add = var:right_extreme_var
						divide  = 1000
					}
				}
			}
			set_global_variable = {
				name = state_ownership_war
				value = 0
			}
			set_global_variable = {
				name = building_numbers_var
				value = 0
			}
			every_scope_building = {
				set_global_variable = {
					name = building_crash_var
					value = private_ownership_fraction
				}
				set_global_variable = {
					name = state_ownership_build
					value = 1
				}
				change_global_variable = {
					name = state_ownership_build
					subtract = private_ownership_fraction
				}
				change_global_variable = {
					name = state_ownership_war
					add = global_var:state_ownership_build
				}
				change_global_variable = {
					name = building_numbers_var
					add = 1
				}
				remove_modifier = HRP_building_crash
				add_modifier = {
					name = HRP_building_crash
					multiplier = {
						add = 1000
						subtract = var:recovery_var
						divide  = 1000
						multiply = global_var:building_crash_var
					}
					months = 2
				}				
			}
			change_global_variable = {
				name = state_ownership_war
				divide = global_var:building_numbers_var
			}
			set_variable = {
				name = state_ownership_loc_var
				value = global_var:state_ownership_war
			}
			every_political_movement = {
				limit = {
					OR = {
						is_political_movement_type = movement_communist
						is_political_movement_type = movement_làngist
						is_political_movement_type = movement_anarchist
						is_political_movement_type = movement_communalist
						is_political_movement_type = movement_edenist
					}
				}
				remove_modifier = HRP_left_extreme_modifier_movement 
				add_modifier = {
					name = HRP_left_extreme_modifier_movement 
					multiplier = {
						add = var:left_extreme_var
						divide = 1000
					}
				}
			}
			every_political_movement = {
				limit = {
					OR = {
						is_political_movement_type = movement_fascist
						is_political_movement_type = movement_eugenist
						is_political_movement_type = movement_prozelyte_corportatist
						is_political_movement_type = movement_national_syndicalist
						is_political_movement_type = movement_anti_western
					}
				}
				remove_modifier = HRP_right_extreme_modifier_movement
				add_modifier = {
					name = HRP_right_extreme_modifier_movement 
					multiplier = {
						add = var:right_extreme_var
						divide = 1000
					}
				}
			}
			set_variable = { 
				name = lowered_gdp
				value = GDP
			}
			change_variable = { 
				name = lowered_gdp
				divide = 100
			}
			change_variable = { 
				name = lowered_gdp
				add = 100000
			}
			set_variable = { 
				name = lowered_gdp_to_tage
				value = var:lowered_gdp
			}
			change_variable = { 
				name = lowered_gdp_to_tage
				divide = -2
			}
			if = {
				limit = {
					var:right_extreme_var >= 250
					NOT = { has_variable = right_agitator_var }
				}
				if = {
					limit = {
						has_modifier = HRP_national_unionist_modifier
					}
					create_character = {
						is_agitator = yes 
						ideology = ideology_national_unionist
					}
				}
				else_if = {
					limit = {
						has_modifier = HRP_eugenist_modifier
					}
					create_character = {
						is_agitator = yes 
						ideology = ideology_eugenist
					}
				}
				else_if = {
					limit = {
						has_modifier = HRP_prozelyte_corportatist_modifier
					}
					create_character = {
						is_agitator = yes 
						ideology = ideology_prozelyte_corporatist
					}
				}
				else_if = {
					limit = {
						has_modifier = HRP_national_syndicalist_modifier
					}
					create_character = {
						is_agitator = yes 
						ideology = ideology_national_syndicalist
					}
				}
				else_if = {
					limit = {
						has_modifier = HRP_anti_western_modifier
					}
					create_character = {
						is_agitator = yes 
						ideology = ideology_anti_western
					}
				}
				set_variable = right_agitator_var
			}
			if = {
				limit = {
					var:left_extreme_var >= 250
					NOT = { has_variable = left_agitator_var }
				}
				if = {
					limit = {
						has_modifier = HRP_maximist_modifier
					}
					create_character = {
						is_agitator = yes 
						ideology = ideology_maximism
					}
				}
				else_if = {
					limit = {
						has_modifier = ideology_làngist
					}
					create_character = {
						is_agitator = yes 
						ideology = HRP_làngist_modifier
					}
				}
				else_if = {
					limit = {
						has_modifier = HRP_anarchistt_modifier
					}
					create_character = {
						is_agitator = yes 
						ideology = ideology_anarchist
					}
				}
				else_if = {
					limit = {
						has_modifier = HRP_communalist_modifier
					}
					create_character = {
						is_agitator = yes 
						ideology = ideology_communalism
					}
				}
				else_if = {
					limit = {
						has_modifier = HRP_edenist_modifier
					}
					create_character = {
						is_agitator = yes 
						ideology = ideology_edenist
					}
				}
				set_variable = left_agitator_var
			}
		}
	}

	on_complete = {
		custom_tooltip = {
			text = "We'll lose all effect of the crash"
			every_scope_building = {
				remove_modifier = HRP_building_crash	
			}
		}
		add_loyalists = {
			value = 0.20
		}
	}
	
	should_be_pinned_by_default = yes
}
