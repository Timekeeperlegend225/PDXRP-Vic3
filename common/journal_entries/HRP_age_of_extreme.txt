je_HRP_recovery_and_extremism = {
	icon = "gfx/interface/icons/generic_icons/company_prosperity.dds"

	group = je_group_historical_content

	scripted_progress_bar = HRP_AoE_recovery
	scripted_progress_bar = HRP_AoE_Left_E
	scripted_progress_bar = HRP_AoE_Right_E

	immediate = {
		set_variable = {
			name = starting_recovery
			value = 1
		}
		change_variable = {
			name = starting_recovery
			subtract = var:national_integration_var
		}
		change_variable = {
			name = starting_recovery
			multiply = 1000
		}
		set_variable = recoverysetup 
		remove_modifier = HRP_crash
		scope:journal_entry = { 
			remove_modifier = HRP_crash
			add_modifier = {
				name = HRP_crash
				multiplier = {
					add = 900
					subtract = var:starting_recovery
					divide  = 1000
				}
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = "Our recovery is complete"
			scope:journal_entry = {
				"scripted_bar_progress(HRP_AoE_recovery)" >= 1000
			}
		}
	}
	
    invalid = {
		custom_tooltip = {
			text = "Age of Extreme is finished"
			NOT = {
				has_global_variable = age_of_extreme_gm
			}
		}
    }
	
	scripted_button = AoE_stock_action
	scripted_button = AoE_pause_interest
	scripted_button = AoE_inflate_currency
	scripted_button = AoE_discourage_free_market
	scripted_button = AoE_socialized_cooperation
	scripted_button = AoE_national_companies

	status_desc = {
	}

	on_monthly_pulse = {
		effect = {
			remove_variable = recoverysetup
			HRP_AoE_recovery_progress = yes 
			HRP_AoE_Left_E_progress = yes
			HRP_AoE_Right_E_progress = yes
			set_variable = { 
				name = recovery_var 
				value = "scope:journal_entry.scripted_bar_progress(HRP_AoE_recovery)" 
			}
			remove_modifier = HRP_stability_visual_modifier
			add_modifier = {
				name = HRP_stability_visual_modifier
				multiplier = var:tempnationalstab
			}
			set_variable = { 
				name = left_extreme_var
				value = "scope:journal_entry.scripted_bar_progress(HRP_AoE_Left_E)" 
			}
			set_variable = { 
				name = right_extreme_var 
				value = "scope:journal_entry.scripted_bar_progress(HRP_AoE_Right_E)" 
			}
			set_variable = { 
				name = both_extreme_var
				value = var:left_extreme_var
			}
			change_variable = { 
				name = both_extreme_var
				add = var:right_extreme_var
			}
			scope:journal_entry = { 
				remove_modifier = HRP_crash
				remove_modifier = HRP_left_extreme_modifier
				remove_modifier = HRP_right_extreme_modifier
				add_modifier = {
					name = HRP_crash
					multiplier = {
						add = 1000
						subtract = var:recovery_var
						divide  = 1000
					}
				}
				add_modifier = {
					name = HRP_left_extreme_modifier
					multiplier = {
						add = var:left_extreme_var
						divide  = 1000
					}
				}
				add_modifier = {
					name = HRP_right_extreme_modifier
					multiplier = {
						add = var:right_extreme_var
						divide  = 1000
					}
				}
			}
			set_global_variable = {
				name = state_ownership_war
				value = 0
			}
			set_global_variable = {
				name = building_numbers_var
				value = 0
			}
			every_scope_building = {
				set_global_variable = {
					name = building_crash_var
					value = private_ownership_fraction
				}
				set_global_variable = {
					name = state_ownership_build
					value = 1
				}
				change_global_variable = {
					name = state_ownership_build
					subtract = private_ownership_fraction
				}
				change_global_variable = {
					name = state_ownership_war
					add = global_var:state_ownership_build
				}
				change_global_variable = {
					name = building_numbers_var
					add = 1
				}
				remove_modifier = HRP_building_crash
				add_modifier = {
					name = HRP_building_crash
					multiplier = {
						add = 1000
						subtract = var:recovery_var
						divide  = 1000
						multiply = global_var:building_crash_var
					}
					months = 2
				}	
			}
			change_global_variable = {
				name = state_ownership_war
				divide = global_var:building_numbers_var
			}
			set_variable = {
				name = state_ownership_loc_var
				value = global_var:state_ownership_war
			}
			every_political_movement = {
				limit = {
					OR = {
						is_political_movement_type = movement_socialist
						is_political_movement_type = movement_communist
						is_political_movement_type = movement_socialist
					}
				}
				add_modifier = {
					name = edenist_popular_support_modifier 
					multiplier = -1
				}
			}
			every_political_movement = {
				limit = {
					OR = {
						is_political_movement_type = movement_bonapartist
					}
				}
				add_modifier = {
					name = edenist_popular_support_modifier 
					multiplier = -1
				}
			}
			if = {
				limit = { 
					has_modifier = game_master
				}
				change_variable = {
					name = GM_crash_countwodn
					add = 1
				}
			}
		}
	}

	on_complete = {
		custom_tooltip = {
			text = "We'll lose all effect of the crash"
			every_scope_building = {
				remove_modifier = HRP_building_crash	
			}
		}
		add_loyalists = {
			value = 0.20
		}
	}
	
	should_be_pinned_by_default = yes
}
