je_GDP_ownership = {
    icon = gfx/interface/icons/generic_icons/gdp.dds

	group = je_group_historical_content

	is_shown_when_inactive = {
		always = yes
	}

	possible = {
		always = yes
	}	
	
	custom_on_completion_header = je_GDP_ownership_on_completion_header
	custom_on_failure_header = je_GDP_ownership_on_fail_header

	on_complete = {
		add_modifier = low_ownership_modifier
	}
	
	on_fail = {
		add_modifier = high_foreign_modifier
	}

	on_weekly_pulse = {
		effect = {
			scope:journal_entry = { 
				remove_modifier = low_ownership_modifier
				remove_modifier = high_foreign_modifier
			}
			set_variable = {
				name = self_gdp_ownership_var
				value = "this.gdp_ownership_ratio(this)"
			}
			set_variable = {
				name = self_gdp_ownership_modifier_temp
				value = 1
			}
			change_variable = {
				name = self_gdp_ownership_modifier_temp
				subtract = 0.25
			}
			change_variable = {
				name = self_gdp_ownership_modifier_temp
				subtract = var:self_gdp_ownership_var
			}
			change_variable = {
				name = self_gdp_ownership_modifier_temp
				multiply = 1.33333
			}
			change_variable = {
				name = self_gdp_ownership_modifier_temp
				multiply = 1.33333
			}
			set_global_variable = {
				name = foreign_gdp_ownership_var
				value = 0
			}
			every_country = {
				limit = { this != prev }
				save_temporary_scope_as = t
				if = {
					limit = { "scope:t.gdp_ownership_ratio(prev)" > 0 }
					set_variable = {
						name = other_gdp_ownership_var
						value = "this.gdp_ownership_ratio(this)"
					}
					change_variable = {
						name = other_gdp_ownership_var
						multiply = scope:t.gdp
					}
					change_global_variable = {
						name = foreign_gdp_ownership_var
						add = var:other_gdp_ownership_var
					}
				}
			}
			set_variable = {
				name = foreign_gdp_ownership_var_temp1
				value = global_var:foreign_gdp_ownership_var
			}
			change_variable = {
				name = foreign_gdp_ownership_var_temp1
				divide = gdp
			}
			set_variable = {
				name = foreign_gdp_ownership_var_temp
				value = var:foreign_gdp_ownership_var_temp1
			}
			change_variable = {
				name = foreign_gdp_ownership_var_temp
				subtract = 0.30
			}
			change_variable = {
				name = foreign_gdp_ownership_var_temp
				divide = 2
			}
			clamp_variable = {
				name = foreign_gdp_ownership_var_temp
				min = 0
				max = 1
			}
			if = {
				limit = {
					var:self_gdp_ownership_modifier_temp > 0
				}
				scope:journal_entry = { 
					add_modifier = {
						name = low_ownership_modifier
						multiplier = {
							add = var:self_gdp_ownership_modifier_temp
						}
					}
				}
			}
			if = {
				limit = {
					var:foreign_gdp_ownership_var_temp > 0
				}
				scope:journal_entry = { 
					add_modifier = {
						name = high_foreign_modifier
						multiplier = {
							add = var:foreign_gdp_ownership_var_temp
						}
					}
				}
			}
		}
	}
	
	should_be_pinned_by_default = yes
}
