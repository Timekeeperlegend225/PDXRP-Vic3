REPLACE_OR_CREATE:pan_plague = {
	trigger = {
		always = yes
	}

	range = {
		integer_range = {
			min = 0
			max = 0
		}
	}

	duration = {
		fixed_range = {
			min = 36
			max = 96
		}
	}

	modifier = {
		state_mortality_mult = 1
		state_standard_of_living_add = -1
	}

	intensity = {
		fixed_range = {
			min = 0
			max = 1
		}
		add = global_var:plague_intesity_var
	}

	chance = {
		value = 0.1
		if = {
			limit = {
				any_neighbouring_state = {
					state_region = {
						has_harvest_condition = pan_plague
					}
				}
			}
			add = 1
		}
				
		subtract = var:average_sr_res_var
		add = var:total_sr_urban_var
		add = var:total_sr_trade_var
		
		if = {
			limit = {
				owner = {
					has_law = law_type:law_no_migration_controls
				}
			}
			add = 0.1
		}
		else_if = {
			limit = {
				owner = {
					has_law = law_type:law_closed_borders
					NOT = {
						any_scope_state = {
							state_region = {
								has_harvest_condition = pan_plague
							}
						}
					}
				}
			}
			multiply = 0.7
		}
		if = {
			limit = {
				owner = {
					has_law = law_type:law_free_trade
				}
			}
			add = 0.1
		}
		else_if = {
			limit = {
				owner = {
					has_law = law_type:law_isolationism
				}
				NOT = {
					any_scope_state = {
						is_in_same_market_area = yes
						state_region = {
							has_harvest_condition = pan_plague
						}
					}
				}
			}
			multiply = 0.7
		}
		if = {
			limit = {
				owner = {
					capital = {
						any_scope_building = {
							has_active_production_method = pm_pan_plague_prevention
						}
					}
				}
			}
			multiply = 0.9
		}
		if = {
			limit = {
				owner = {
					OR = {
						has_law = law_type:law_traditionalism
						has_law = law_type:law_agrarianism
					}
				}
			}
			multiply = 0.95
		}
		else_if = {
			limit = {
				owner = {
					has_law = law_type:law_industry_banned
				}
			}
			multiply = 0.90
		}
	}

	color = { 100 0 0 }
	icon = "gfx/interface/icons/harvest_condition_icons/plague.dds"
	map_texture = "gfx/map/map_painting/harvest_conditions_map_patterns/harvest_conditions_bad.dds"
	graphics = disease_outbreak
}
