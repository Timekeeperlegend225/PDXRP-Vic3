pdxrp_start_world_soul = {
    is_shown = {
        c:MOA ?= THIS
        NOT = { has_global_variable = pdxrp_world_soul_temperature }
    }
    possible = { always = yes }
    when_taken = {
        custom_tooltip = pdxrp_start_world_soul_tt
        hidden_effect = { pdxrp_world_soul_start_effect = yes }
    }
    ai_chance = { value = 0 }
}

pdxrp_open_chrome = {
    is_shown = {
        c:MOA ?= THIS
        has_global_variable = pdxrp_world_soul_temperature
    }
    possible = { always = yes }
    when_taken = {
        custom_tooltip = pdxrp_open_chrome_tt
        hidden_effect = {
            pdxrp_world_soul_clamp_temperature_effect = yes
            change_global_variable = { name = pdxrp_world_soul_temperature add = 3 }
            pdxrp_world_soul_sync_journals_effect = yes
        }
    }
    ai_chance = { value = 0 }
}

pdxrp_open_window = {
    is_shown = {
        c:MOA ?= THIS
        has_global_variable = pdxrp_world_soul_temperature
    }
    possible = { global_var:pdxrp_world_soul_temperature > 0 }
    when_taken = {
        custom_tooltip = pdxrp_open_window_tt
        hidden_effect = {
            pdxrp_world_soul_clamp_temperature_effect = yes
            change_global_variable = { name = pdxrp_world_soul_temperature subtract = 3 }
            pdxrp_world_soul_sync_journals_effect = yes
        }
    }
    ai_chance = { value = 0 }
}
