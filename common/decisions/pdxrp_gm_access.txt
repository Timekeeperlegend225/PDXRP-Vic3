pdxrp_monaco_remove_other_game_masters = {
    is_shown = {
        is_player = yes
        pdxrp_is_campaign_host = yes
    }
    possible = {
        any_country = {
            NOT = { THIS = ROOT }
            has_modifier = GM_game_master
        }
    }
    when_taken = {
        every_country = {
            limit = {
                NOT = { THIS = ROOT }
                has_modifier = GM_game_master
            }
            remove_modifier = GM_game_master
        }
    }
    ai_chance = { value = 0 }
}
